"""Run the unchanged Cargo graph, then freeze and execute exact new test binaries.

No production or whole-workspace qualification follows from this scoped runner.
"""
from pathlib import Path
import hashlib
import json
import os
import shutil
import subprocess
import time
import uuid

root = Path(__file__).resolve().parent.parent
source = root / 'pq-base5e'
evidence = root / 'evidence'
toolchain = root.parent / 'pon-build-inputs/unpacked/toolchain/bin'
label = 'merged-native-' + uuid.uuid4().hex[:12]
logpath = evidence / (label + '.log')

def git(*args):
    return subprocess.check_output(['git', *args], cwd=source, text=True).strip()

def hash_file(path):
    h = hashlib.sha256()
    with path.open('rb') as stream:
        for data in iter(lambda: stream.read(1024 * 1024), b''):
            h.update(data)
    return h.hexdigest()

def verify_source():
    head = git('rev-parse', 'HEAD')
    count = 0
    for line in git('ls-tree', '-r', 'HEAD').splitlines():
        meta, name = line.split('\t', 1)
        if name.startswith('evidence/'):
            continue
        mode, kind, wanted = meta.split()
        path = source / name
        assert kind == 'blob' and path.is_file() and not path.is_symlink(), name
        raw = path.read_bytes()
        actual = hashlib.sha1(b'blob ' + str(len(raw)).encode() + b'\0' + raw).hexdigest()
        assert actual == wanted, (name, actual, wanted)
        assert bool(path.stat().st_mode & 0o111) == (mode == '100755'), name
        count += 1
    assert not git('status', '--porcelain', '--untracked-files=all')
    return {'head': head, 'tree': git('rev-parse', 'HEAD^{tree}'),
            'non_evidence_blobs_verified': count, 'historical_evidence_sparse': True,
            'full_repository_source_guard_passed': False}

before = verify_source()
env = dict(os.environ)
env.update(PATH=str(toolchain) + os.pathsep + env.get('PATH', ''),
           RUSTC=str(toolchain / 'rustc'),
           CARGO_HOME=str(root.parent / 'pon-build-inputs/native-cargo-home'),
           CARGO_TARGET_DIR=str(root / 'target-native'), CARGO_BUILD_JOBS='2', RUST_TEST_THREADS='1')
command = [str(toolchain / 'cargo'), 'test', '--offline', '--locked', '--message-format=json',
           '-p', 'trnm-protocol', '-p', 'trnm-mvcc-fee', '-p', 'trnm-crypto-primitives',
           '-p', 'trnm-pon-node', '--all-features', '--lib', 'pq_']
started = time.monotonic()
with logpath.open('w') as log:
    build = subprocess.run(command, cwd=source / 'trillionnium', env=env, stdout=log, stderr=subprocess.STDOUT)
result = {'label': label, 'source': before, 'command': command,
          'rustc': subprocess.check_output([str(toolchain / 'rustc'), '-Vv'], text=True),
          'cargo_exit': build.returncode, 'cargo_log': str(logpath), 'binaries': [],
          'production_activation': False, 'native_pq_node_integrated': False,
          'full_workspace_tests_executed': False, 'quantum_pon_hardness_qualified': False}
expected = {
    'trnm_protocol': 'pq_transfer::tests::alleged_signature_is_not_parser_authority_or_transaction_identity',
    'trnm_mvcc_fee': 'pon_executor::pq_transfer::tests::real_pq_signature_reaches_existing_native_nonce_fee_and_transfer_rules',
    'trnm_crypto_primitives': 'pq_auth::tests::complete_pinned_wycheproof_corpus_including_duplicate_hints',
    'trnm_pon_node': 'pq_authority::tests::real_sqlite_full_failure_rolls_back_sequence_and_record_together',
}
artifacts = {}
for line in logpath.read_text().splitlines():
    try:
        event = json.loads(line)
    except ValueError:
        continue
    if event.get('reason') == 'compiler-artifact' and event.get('profile', {}).get('test'):
        name = event.get('target', {}).get('name')
        if name in expected and event.get('executable'):
            artifacts[name] = Path(event['executable'])
if build.returncode == 0:
    assert set(artifacts) == set(expected), artifacts
    frozen = evidence / 'frozen-binaries'
    frozen.mkdir(exist_ok=True)
    for name, original in sorted(artifacts.items()):
        digest = hash_file(original)
        copy = frozen / (label + '-' + name + '-' + digest[:16])
        with original.open('rb') as incoming, copy.open('xb') as outgoing:
            shutil.copyfileobj(incoming, outgoing)
        copy.chmod(0o755)
        assert digest == hash_file(copy) == hash_file(original)
        listing = subprocess.check_output([str(copy), '--list'], text=True)
        assert expected[name] + ': test' in listing, (name, listing)
        testlog = evidence / (label + '-' + name + '-frozen-tests.log')
        selector = [] if name in {'trnm_protocol', 'trnm_mvcc_fee'} else ['pq_']
        with testlog.open('w') as log:
            run = subprocess.run([str(copy), *selector, '--test-threads=1'], env=env,
                                 cwd=source / 'trillionnium', stdout=log, stderr=subprocess.STDOUT)
        assert digest == hash_file(copy)
        result['binaries'].append({'package': name, 'binary': str(copy), 'sha256': digest,
                                   'list_sha256': hashlib.sha256(listing.encode()).hexdigest(),
                                   'required_new_test_present': True, 'selector': selector,
                                   'exit': run.returncode, 'log': str(testlog),
                                   'log_sha256': hash_file(testlog)})
        if run.returncode:
            break
result['source_after'] = verify_source()
assert result['source_after'] == before
result['elapsed_seconds'] = time.monotonic() - started
result['cargo_log_sha256'] = hash_file(logpath)
result['success'] = (build.returncode == 0 and len(result['binaries']) == len(expected)
                     and all(b['exit'] == 0 for b in result['binaries']))
output = evidence / (label + '.json')
output.write_text(json.dumps(result, indent=2) + '\n')
print(json.dumps({'label': label, 'result': str(output), 'cargo_exit': build.returncode,
                  'success': result['success'], 'binaries': len(result['binaries']),
                  'elapsed_seconds': result['elapsed_seconds']}), flush=True)
raise SystemExit(0 if result['success'] else 1)
