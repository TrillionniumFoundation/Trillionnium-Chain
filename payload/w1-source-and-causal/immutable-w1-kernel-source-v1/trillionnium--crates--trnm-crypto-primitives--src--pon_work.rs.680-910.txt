709:fn checked_proof_statement(
710:    challenge: Hash,
711:    expected_task: Hash,
712:    target: Hash,
713:    bytes: &[u8],
714:) -> Result<CheckedProofStatement<'_>, WorkError> {
715:    if bytes.len() != PROOF_BYTES {
716:        return Err(WorkError::Length);
717:    }
718:    if &bytes[..4] != b"PNW1" {
719:        return Err(WorkError::Version);
720:    }
721:    if target == [0; 32] {
722:        return Err(WorkError::Target);
723:    }
724:    // Preserve C, A, B field-error precedence before task, ticket and
725:    // callbacks. Exact fixed slices need no allocation, unaligned cast,
726:    // host-endian assumption or reduction of noncanonical input words.
727:    for matrix in [2, 0, 1] {
728:        let start = 4 + matrix * CELLS * 4;
729:        if bytes[start..start + CELLS * 4]
730:            .chunks_exact(4)
731:            .any(|p| u128::from(u32::from_le_bytes([p[0], p[1], p[2], p[3]])) >= Q)
732:        {
733:            return Err(WorkError::Field);
734:        }
735:    }
736:    // These are the current canonical bytes, not a cached task verdict.
737:    let actual_task = hash(
738:        b"task",
739:        &[
740:            &bytes[4..4 + CELLS * 4],
741:            &bytes[4 + CELLS * 4..4 + CELLS * 8],
742:        ],
743:    );
744:    if actual_task != expected_task {
745:        return Err(WorkError::Task);
746:    }
747:    let trace_claim: &[u8] = &bytes[4 + 3 * CELLS * 4..];
748:    // Cheap ticket rejection before expensive transcript verification is only an admission filter.
749:    // A passing filter alone never constructs VerifiedWork.
750:    let ticket = hash(b"ticket", &[&challenge, trace_claim]);
751:    if ticket > target {
752:        return Err(WorkError::Target);
753:    }
754:    Ok(CheckedProofStatement {
755:        inputs: &bytes[4..4 + 2 * CELLS * 4],

757:        ticket,
758:    })
