185: impl HistoryImportObserver {
186:     pub fn new() -> Self {
187:         let epoch = Instant::now();
188:         Self {
189:             epoch,
190:             page: TaskObservation::local(epoch),
191:             packets: std::array::from_fn(|_| TaskObservation::local(epoch)),
192:             active_packet: AtomicUsize::new(MAX_HISTORY_IMPORT_OBSERVATION_PACKETS),
193:             context: Mutex::new(Context::default()),
194:             failures: AtomicU64::new(0),
195:             host: Mutex::new(HostRecords::new()),
196:             page_attribution: NativeAttribution::new(epoch),
197:             packet_attribution: std::array::from_fn(|_| NativeAttribution::new(epoch)),
198:         }
199:     }
200:     fn fail(&self) {
201:         let _ = self
202:             .failures
203:             .fetch_update(Ordering::Relaxed, Ordering::Relaxed, |n| {
204:                 Some(n.saturating_add(1))
205:             });
206:     }
207:     pub(super) fn enter(&self, tip: Hash, after: Hash, packets: usize) {
208:         match self.context.try_lock() {
209:             Ok(mut context) if !context.entered => {
210:                 context.entered = true;
211:                 context.tip = Some(tip);
212:                 context.after = Some(after);
213:                 context.supplied_packets = Some(packets);
214:             }
215:             _ => self.fail(),
216:         }
217:     }
218:     pub(super) fn page(&self) -> &TaskObservation {
219:         &self.page
220:     }
221:     pub(super) fn native_page(&self) -> &NativeAttribution {
222:         &self.page_attribution
223:     }
224:     pub(super) fn native_packet(&self, index: usize) -> Option<&NativeAttribution> {
225:         let record = self.packet_attribution.get(index);
226:         if record.is_none() {
227:             self.fail();
228:         }
229:         record
230:     }
231:     pub(super) fn decoded_packet(&self, index: usize, id: Hash) {
232:         match self.context.try_lock() {
233:             Ok(mut context) if index < context.packet_ids.len() => {
234:                 if context.packet_ids[index].is_some() {
235:                     self.fail();
236:                 } else {
237:                     context.packet_ids[index] = Some(id);
238:                 }
239:             }
240:             _ => self.fail(),
241:         }
242:     }
243:     pub(super) fn packet(&self, index: usize) -> Option<&TaskObservation> {
244:         let packet = self.packets.get(index);
245:         if packet.is_some() {
246:             self.active_packet.store(index, Ordering::Release);
247:         } else {
248:             self.fail();
249:         }
250:         packet
251:     }
252: 
253:     fn checkpoint_phase(&self) -> usize {
254:         let observed = (|| {
255:             let index = self.active_packet.load(Ordering::Acquire);
256:             if let Some(packet) = self.packets.get(index) {
257:                 if let Some(phase) = packet.active_phase()? {
258:                     return Ok(Some(phase));
259:                 }
260:             }
261:             self.page.active_phase()
262:         })();
263:         match observed {
264:             Ok(Some(phase)) => phase as usize,
265:             Ok(None) => UNATTRIBUTED_HOST_PHASE,
266:             Err(()) => {
267:                 self.fail();
268:                 UNATTRIBUTED_HOST_PHASE
269:             }
270:         }
271:     }
272:     /// Snapshot after the synchronous import returned/unwound. A live snapshot
273:     /// can contain unfinished spans and does not establish service completion.
274:     pub fn snapshot(&self) -> HistoryImportObservationSnapshot {
275:         let (host_checks, checkpoints) = match self.host.lock() {
276:             Ok(host) => (host.phases, host.checkpoints),
277:             Err(_) => {
278:                 self.fail();
279:                 let host = HostRecords::new();
280:                 (host.phases, host.checkpoints)
281:             }
282:         };
283:         let context = match self.context.lock() {
284:             Ok(context) => *context,
285:             Err(_) => {
286:                 self.fail();
287:                 Context::default()
288:             }
289:         };
290:         let (phases, phase_failures) = self.page.phase_snapshot();
291:         let (page_attribution, attribution_failures) = self.page_attribution.snapshot();
292:         let mut failures = self
293:             .failures
294:             .load(Ordering::Acquire)
295:             .saturating_add(phase_failures)
296:             .saturating_add(attribution_failures);
297:         let mut packet_attributions = Vec::new();
298:         let packets = self
299:             .packets
300:             .iter()
301:             .enumerate()
302:             .filter_map(|(index, packet)| {
303:                 let (phases, phase_failures) = packet.phase_snapshot();
304:                 let (native_attribution, attribution_failures) =
305:                     self.packet_attribution[index].snapshot();
306:                 failures = failures
307:                     .saturating_add(phase_failures)
308:                     .saturating_add(attribution_failures);
309:                 if let Some(attribution) = native_attribution {
310:                     packet_attributions.push(HistoryPacketNativeAttribution {
311:                         packet_index: index,
312:                         packet_id: context.packet_ids[index],
313:                         attribution,
314:                     });
315:                 }
316:                 (context.packet_ids[index].is_some() || phases.iter().any(|phase| phase.started))
317:                     .then_some(HistoryPacketObservation {
318:                         packet_index: index,
319:                         packet_id: context.packet_ids[index],
320:                         phases,
321:                     })
322:             })
323:             .collect();
324:         let native_attribution = (page_attribution.is_some() || !packet_attributions.is_empty())
325:             .then_some(HistoryImportNativeAttribution {
326:                 schema: "pon-local-history-import-native-attribution-v1",
327:                 page: page_attribution,
328:                 packets: packet_attributions,
329:             });
330:         HistoryImportObservationSnapshot {
331:             schema: "pon-local-history-import-observation-v1",
332:             tip: context.tip,
333:             after: context.after,
334:             supplied_packets: context.supplied_packets,
335:             packet_capacity: MAX_HISTORY_IMPORT_OBSERVATION_PACKETS,
336:             phases,
337:             native_attribution,
338:             packets,
339:             measurement_failures: failures,
340:             host_checks,
341:             host_checkpoint_attribution: HistoryHostCheckpointAttribution {
342:                 schema: "pon-local-history-host-phase-attribution-v1",
343:                 buckets: std::array::from_fn(|index| HistoryHostCheckpointPhase {
344:                     owner_phase: SubmitPhase::ALL.get(index).copied(),
345:                     calls: checkpoints[index],
346:                 }),
347:                 all_checkpoint_calls_attributed: failures == 0
348:                     && checkpoints[UNATTRIBUTED_HOST_PHASE].calls_started == 0
349:                     && checkpoints.iter().try_fold(0u64, |sum, row| {
350:                         sum.checked_add(row.calls_started)
351:                     }) == Some(host_checks[HistoryHostPhase::Checkpoint as usize].calls_started),
352:                 has_consensus_authority: false,
353:                 scope: "actual Host checkpoint calls classified once at entry by the innermost open owner phase; worker calls use the current owner phase context; absent or ambiguous contexts remain unattributed; the same existing call clocks feed aggregate and bucket records, which overlap; summed call wall intervals are not elapsed runtime and may overlap across threads; Host CPU includes each invoking thread, while native phase CPU excludes scoped workers; observer overhead remains in outer measured work",
354:             },
355:             host_checks_are_nested_not_additive: true,
356:             cpu_intervals_are_nested_not_additive: true,
357:             phase_cpu_excludes_scoped_workers: true,
358:             host_cpu_includes_each_calling_thread: true,
359:             host_cpu_is_full_worker_cpu: false,
360:             observation_has_consensus_authority: false,
361:             clock_scope: "one observer-local monotonic epoch; page import includes packet and checkpoint Host spans; phase CPU is importer thread only; Host CPU sums each actual invoking thread's Host intervals, not complete worker CPU; resolve/initial/final Host spans are outside native import but inside client replay; network fetch is outside client replay",
362:         }
363:     }
364: }

366: fn add_sample(total: Option<u64>, sample: Option<u64>, first: bool) -> Option<u64> {
367:     if first {
368:         sample
369:     } else {
370:         total
371:             .zip(sample)
372:             .and_then(|(sum, next)| sum.checked_add(next))
373:     }

376: struct HostGuard<'a> {
377:     observer: &'a HistoryImportObserver,
378:     phase: HistoryHostPhase,
379:     checkpoint_phase: Option<usize>,
380:     start: Instant,
381:     cpu: Option<ThreadCpuStamp>,
382:     accepted: Option<bool>,
383:     error_code: Option<&'static str>,
384:     recorded: bool,
385: }
386: impl<'a> HostGuard<'a> {
387:     fn new(observer: &'a HistoryImportObserver, phase: HistoryHostPhase) -> Self {
388:         // This is an observation of existing spans, never another Host action.
389:         // It is lock-free and its overhead remains in the outer native span.
390:         let mut checkpoint_phase =
391:             matches!(phase, HistoryHostPhase::Checkpoint).then(|| observer.checkpoint_phase());
392:         let start = Instant::now();
393:         let offset = start
394:             .checked_duration_since(observer.epoch)
395:             .and_then(|elapsed| u64::try_from(elapsed.as_nanos()).ok());
396:         let mut recorded = false;
397:         let mut checkpoint_recorded = false;
398:         if let Ok(mut host) = observer.host.try_lock() {
399:             let HostRecords {
400:                 phases,
401:                 checkpoints,
402:             } = &mut *host;
403:             let rows = [
404:                 Some(&mut phases[phase as usize]),
405:                 checkpoint_phase.map(|index| &mut checkpoints[index]),
406:             ];
407:             for (index, row) in rows.into_iter().enumerate() {
408:                 let Some(row) = row else { continue };
409:                 if let Some(count) = row.calls_started.checked_add(1) {
410:                     row.first_started_ns = if count == 1 {
411:                         offset
412:                     } else {
413:                         row.first_started_ns
414:                             .zip(offset)
415:                             .map(|(before, now)| before.min(now))
416:                     };
417:                     row.calls_started = count;
418:                     if index == 0 {
419:                         recorded = true;
420:                     } else {
421:                         checkpoint_recorded = true;
422:                     }
423:                 } else {
424:                     observer.fail();
425:                 }
426:             }
427:         } else {
428:             observer.fail();
429:         }
430:         if !checkpoint_recorded {
431:             checkpoint_phase = None;
432:         }
433:         Self {
434:             observer,
435:             phase,
436:             checkpoint_phase,
437:             start,
438:             cpu: ThreadCpuStamp::start(),
439:             accepted: None,
440:             error_code: None,
441:             recorded,
442:         }
443:     }
444:     fn finish<T>(mut self, result: &crate::Result<T>) {
445:         self.accepted = Some(result.is_ok());
446:         self.error_code = result
447:             .as_ref()
448:             .err()
449:             .and_then(crate::Error::code)
450:             .map(crate::ErrorCode::as_str);
451:     }
452: }
453: impl Drop for HostGuard<'_> {
454:     fn drop(&mut self) {
455:         if !self.recorded {
456:             return;
457:         }
458:         let end = Instant::now();
459:         let ended = end
460:             .checked_duration_since(self.observer.epoch)
461:             .and_then(|elapsed| u64::try_from(elapsed.as_nanos()).ok());
462:         let wall = end
463:             .checked_duration_since(self.start)
464:             .and_then(|elapsed| u64::try_from(elapsed.as_nanos()).ok());
465:         let cpu = self.cpu.take().and_then(ThreadCpuStamp::finish);
466:         if let Ok(mut host) = self.observer.host.try_lock() {
467:             let HostRecords {
468:                 phases,
469:                 checkpoints,
470:             } = &mut *host;
471:             let rows = [
472:                 Some(&mut phases[self.phase as usize]),
473:                 self.checkpoint_phase.map(|index| &mut checkpoints[index]),
474:             ];
475:             for row in rows.into_iter().flatten() {
476:                 let first = row.calls_completed == 0 && row.calls_unfinished == 0;
477:                 row.wall_ns = add_sample(row.wall_ns, wall, first);
478:                 row.thread_cpu_ns = add_sample(row.thread_cpu_ns, cpu, first);
479:                 row.last_ended_ns = if first {
480:                     ended
481:                 } else {
482:                     row.last_ended_ns
483:                         .zip(ended)
484:                         .map(|(before, now)| before.max(now))
485:                 };
486:                 row.wall_clock_unknown_calls = row
487:                     .wall_clock_unknown_calls
488:                     .saturating_add(u64::from(wall.is_none()));
489:                 row.cpu_clock_unknown_calls = row
490:                     .cpu_clock_unknown_calls
491:                     .saturating_add(u64::from(cpu.is_none()));
492:                 match self.accepted {
493:                     Some(accepted) => {
494:                         row.calls_completed = row.calls_completed.saturating_add(1);
495:                         if accepted {
496:                             row.calls_accepted = row.calls_accepted.saturating_add(1);
497:                         } else {
498:                             row.calls_rejected = row.calls_rejected.saturating_add(1);
499:                             row.last_error_code = self.error_code;
500:                         }
501:                     }
502:                     None => {
503:                         row.calls_unfinished = row.calls_unfinished.saturating_add(1);
504:                     }
505:                 }
506:                 if wall.is_some() && row.wall_ns.is_none() && row.wall_clock_unknown_calls == 0 {
507:                     self.observer.fail(); // Aggregate overflow cannot look like zero work.
508:                 }
509:                 if cpu.is_some() && row.thread_cpu_ns.is_none() && row.cpu_clock_unknown_calls == 0
510:                 {
511:                     self.observer.fail();
512:                 }
513:             }
514:         } else {
515:             self.observer.fail();
516:         }
517:     }
518: }

520: pub(in crate::ingress) fn observe_host<T>(
521:     observer: Option<&HistoryImportObserver>,
522:     phase: HistoryHostPhase,
523:     action: impl FnOnce() -> crate::Result<T>,
524: ) -> crate::Result<T> {
525:     let guard = observer.map(|observer| HostGuard::new(observer, phase));
526:     let result = action();
527:     if let Some(guard) = guard {
528:         guard.finish(&result);
529:     }
530:     result
531: }
