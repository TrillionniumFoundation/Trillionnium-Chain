fn append_receipt(file: &mut fs::File, record: &Value, deadline: Instant) -> Result<()> {
    before(deadline)?;
    serde_json::to_writer(&mut *file, record)?;
    file.write_all(b"\\n")?;
    file.sync_data()?;
    before(deadline)
}''')
