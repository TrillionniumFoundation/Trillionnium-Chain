2319: // fence. Unlike a service result, Drop also runs on a local error/unwind.

2324:     node: Arc<Mutex<Node>>,
2325:     ordinary: bool,
2326:     finished: bool,

2328: impl OrdinaryPublicServiceCrashFence {
2329:     fn begin(node: Arc<Mutex<Node>>, startup_deadline: Instant, stop: &AtomicBool) -> Result<Self> {
2330:         // This is the same original startup attempt as profile validation.
2331:         // Contention after that first owner scope must not become an unbounded
2332:         // mutex wait or publish a new durable epoch after cancellation.
2333:         let mut progress = |_: u64| public_startup_alive(startup_deadline, stop);
2334:         let mut owner = lock_owner(&node, &mut progress)?;
2335:         let ordinary = owner.begin_ordinary_public_service_epoch()?;
2336:         drop(owner);
2337:         Ok(Self {
2338:             node,
2339:             ordinary,
2340:             finished: false,
2341:         })
2342:     }
2343: 
2344:     fn complete(&mut self, clean: bool) -> Result<()> {
2345:         if self.ordinary {
2346:             self.node
2347:                 .lock()
2348:                 .map_err(|_| "PUBLIC_OWNER")?
2349:                 .finish_ordinary_public_service_epoch(clean)?;
2350:         }
2351:         self.finished = true;
2352:         Ok(())
2353:     }
2354: }

2356:     fn drop(&mut self) {

2358:             // A poisoned owner cannot silently revoke an unknown active
2359:             // service. The durable active marker was committed BEFORE workers.

2361:                 .node
2362:                 .lock()
2363:                 .unwrap_or_else(std::sync::PoisonError::into_inner);
2364:             owner.mark_ordinary_public_service_epoch_unknown();
