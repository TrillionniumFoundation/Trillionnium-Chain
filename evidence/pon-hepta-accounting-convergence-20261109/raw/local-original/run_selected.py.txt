import hashlib,json,os,pathlib,subprocess,sys,time
root=pathlib.Path('/workspace/scratch/d9bbb5ad0229/chain-accounting-convergence')
out=pathlib.Path('/workspace/scratch/d9bbb5ad0229/accounting-convergence-setup/results')
out.mkdir(exist_ok=True)
sha=subprocess.check_output(['git','rev-parse','HEAD'],cwd=root,text=True).strip()
def snapshot():
 rows=subprocess.check_output(['git','ls-tree','-r','-z','HEAD'],cwd=root).split(b'\0')
 checked=missing=0; manifest=[]
 for row in rows:
  if not row:continue
  meta,path=row.split(b'\t',1);mode,kind,blob=meta.decode().split();rel=path.decode();p=root/rel
  if kind!='blob':continue
  if not p.exists():
   if not rel.startswith('evidence/'):raise RuntimeError('missing non-evidence input '+rel)
   missing+=1;continue
  raw=p.read_bytes();actual=hashlib.sha1(b'blob '+str(len(raw)).encode()+b'\0'+raw).hexdigest()
  if actual!=blob:raise RuntimeError('source mismatch '+rel)
  checked+=1;manifest.append([rel,blob,len(raw)])
 return {'commit':sha,'materialized_checked':checked,'unmaterialized_historical_evidence':missing,'materialized_manifest_sha256':hashlib.sha256(json.dumps(manifest,separators=(',',':')).encode()).hexdigest(),'whole_materialized_repository_checked':missing==0}
start=snapshot();(out/'source-before.json').write_text(json.dumps(start,indent=2)+'\n')
mods=['test_grouped_fresh_deployment_cost_v0','test_grouped_fanout_reanalysis_v1','test_service_work_settlement_v1','test_grouped_legal_strategy_v0','test_grouped_low_rank_v0']
results=[]
for mod in mods:
 argv=[sys.executable,'-m','unittest','-v',mod];t=time.monotonic()
 env=dict(os.environ,PYTHONDONTWRITEBYTECODE='1')
 run=subprocess.run(argv,cwd=root/'formal/pon-nakamoto-v1',env=env,stdout=subprocess.PIPE,stderr=subprocess.PIPE,timeout=180)
 (out/(mod+'.stdout')).write_bytes(run.stdout);(out/(mod+'.stderr')).write_bytes(run.stderr)
 results.append({'module':mod,'argv':argv,'exit_code':run.returncode,'wall_seconds':time.monotonic()-t,'stdout_sha256':hashlib.sha256(run.stdout).hexdigest(),'stderr_sha256':hashlib.sha256(run.stderr).hexdigest(),'terminal':run.stderr.decode(errors='replace').splitlines()[-4:]})
 print(json.dumps(results[-1]),flush=True)
end=snapshot();(out/'source-after.json').write_text(json.dumps(end,indent=2)+'\n')
summary={'source':sha,'source_before':start,'source_after':end,'results':results,'all_passed':all(r['exit_code']==0 for r in results),'native_rerun':False,'qualification':'HOLD_NO_GO'}
(out/'summary.json').write_text(json.dumps(summary,indent=2)+'\n')
sys.exit(0 if summary['all_passed'] else 1)
