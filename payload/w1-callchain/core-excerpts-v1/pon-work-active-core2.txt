1139: // Eight independent columns keep the right-hand row contiguous. LLVM may

1141: // For each column, p=lo+2^32*hi and 2^32=5 (mod q). Including prior,
1142: // low<9*2^32, high<8*2^32, sum=low+5*high<49*2^32. Its fold is
1143: // <2^32+240<2q, so one subtraction is exact. No u64 sum can overflow.

1147:     debug_assert_eq!(right.len(), (R - 1) * N + R);
1148:     debug_assert_eq!(prior.len(), R);
1149:     let mut low = [0u64; R];
1150:     let mut high = [0u64; R];
1151:     for j in 0..R {
1152:         low[j] = u64::from(prior[j]);
1153:     }
1154:     for k in 0..R {
1155:         let row = &right[k * N..k * N + R];
1156:         let value = u64::from(left[k]);
1157:         for j in 0..R {
1158:             let product = value * u64::from(row[j]);
1159:             low[j] += u64::from(product as u32);
1160:             high[j] += product >> 32;

1182: // Both product corrections have shape (N x R) * (R x N). Subtract each
1183: // complete dot product directly into this call's owned result, avoiding a
1184: // second full-size output matrix. Every original MatrixRow callback still
1185: // precedes that row's arithmetic. On cancellation the caller discards this
1186: // operation-local result; partial updates are never published as a fact.
1187: fn subtract_transposed_correction_with_progress<const LIMB: bool, E>(
1188:     left: &[u32],
1189:     right: &[u32],
1190:     product: &mut [u32],
1191:     progress: &mut impl FnMut(VerificationProgress) -> Result<(), E>,
1192: ) -> Result<(), E> {
1193:     debug_assert_eq!(left.len(), N * R);
1194:     debug_assert_eq!(right.len(), R * N);
1195:     debug_assert_eq!(product.len(), CELLS);
1196:     let mut transposed = vec![0; right.len()];
1197:     for k in 0..R {
1198:         for j in 0..N {
1199:             transposed[j * R + k] = right[k * N + j];
1200:         }
1201:     }
1202:     for (i, row) in product.chunks_exact_mut(N).enumerate() {
1203:         progress(VerificationProgress::MatrixRow { row: i })?;
1204:         let a = &left[i * R..(i + 1) * R];
1205:         for (j, value) in row.iter_mut().enumerate() {
1206:             let b = &transposed[j * R..(j + 1) * R];
1207:             let correction = if LIMB {
1208:                 maintenance_limb::dot_bounded(a, b, 0)
1209:             } else {
1210:                 let mut sum = 0_u128;
1211:                 for k in 0..R {
1212:                     sum += u128::from(a[k]) * u128::from(b[k]);
1213:                 }
1214:                 producer_reduce(sum)
1215:             };
1216:             // Canonical operands make this nonnegative and below 2Q. Two
1217:             // successive subtractions equal the original (C' - C1 - C2) mod Q.
1218:             let corrected = u64::from(*value) + Q as u64 - u64::from(correction);
1219:             *value = if corrected >= Q as u64 {
1220:                 (corrected - Q as u64) as u32
1221:             } else {
1222:                 corrected as u32
1223:             };
1224:         }
1225:     }
1226:     Ok(())
1227: }
