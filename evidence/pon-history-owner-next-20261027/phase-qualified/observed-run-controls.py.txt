def run(name, selector, *, binary='trnm_pon_node', exact=False, ignored=False, skip=(), limit=180):
    value = BINARIES[binary]
    exe = Path(value['path'])
    assert hashlib.sha256(exe.read_bytes()).hexdigest() == value['sha256']
    folder = OUT / name
    folder.mkdir(exist_ok=False)
    args = [str(exe), selector, *(['--exact'] if exact else []), *(['--ignored'] if ignored else []),
            '--nocapture', '--test-threads=1']
    for item in skip:
        args.extend(['--skip', item])
    row = {'schema': 'history-frozen-native-control-v2', 'source_commit': BUILD['source_commit'],
           'binary_sha256': value['sha256'], 'argv': args, 'outer_observer_timeout_seconds': limit,
           'started_utc': datetime.datetime.now(datetime.timezone.utc).isoformat()}
    start = time.monotonic()
    with (folder / 'stdout.txt').open('xb') as stdout, (folder / 'stderr.txt').open('xb') as stderr:
        child = subprocess.Popen(args, cwd=ROOT, stdout=stdout, stderr=stderr, start_new_session=True)
        try:
            code = child.wait(timeout=limit)
            timed_out = False
        except subprocess.TimeoutExpired:
            os.killpg(child.pid, signal.SIGKILL)
            code = child.wait()
            timed_out = True
    row.update(exit_code=code, elapsed_seconds=time.monotonic() - start, timed_out=timed_out)
    for stream in ('stdout.txt', 'stderr.txt'):
        b = (folder / stream).read_bytes()
        row[stream] = {'bytes': len(b), 'sha256': hashlib.sha256(b).hexdigest()}
    text = (folder / 'stdout.txt').read_text()
    row['harness_results'] = [{'outcome': m[0], 'passed': int(m[1]), 'failed': int(m[2]), 'ignored': int(m[3])}
                              for m in re.findall(r'test result: (\w+)\. (\d+) passed; (\d+) failed; (\d+) ignored;', text)]
    (folder / 'receipt.json').write_text(json.dumps(row, indent=2) + '\n')
    results.append({'group': name, **row})
    (OUT / 'native-controls.json').write_text(json.dumps(results, indent=2) + '\n')
    print(json.dumps({'group': name, 'exit_code': code, 'elapsed_seconds': row['elapsed_seconds'],
                      'harness_results': row['harness_results']}), flush=True)
    return text
