#!/usr/bin/env python3
"""Copy an explicit closed-preparation allowlist; never write candidate source."""
from __future__ import annotations
import datetime as dt
import hashlib
import json
import os
from pathlib import Path
import stat

PREFIX = '/home/qq/work/chain-cell-native-shard-674c72-20261110'
VALIDATION = Path(PREFIX + '-validation')
CONTROL = Path(PREFIX + '-control')
STAGE = VALIDATION / 'evidence-staging-checked-s3-e6-preflight-v1'
SELECTION = CONTROL / 'collect-batch4-v1.selection.json'
INVENTORY = CONTROL / 'collect-batch4-v1.top-inventory.json'
TERMINALS = CONTROL / 'collect-batch4-v1.terminal-inventory.jsonl'
TEXT_SUFFIXES = {'.json', '.jsonl', '.log', '.txt', '.out', '.err', '.stdout',
                 '.stderr', '.md', '.py', '.rs', '.toml', '.lock', '.sh', '.yml',
                 '.yaml', '.csv', '.tsv', '.sha256', '.diff', '.patch'}
TEXT_NAMES = {'stdout', 'stderr', 'Cargo.lock'}
RENAME_SUFFIXES = {'.md', '.py', '.rs', '.toml', '.lock', '.sh', '.yml', '.yaml', '.log'}
EXCLUDED_DIRS = {'binaries', 'tmp', 'target', '.git', 'node_modules', '__pycache__', 'cache',
                 'wire-cache', 'consumer-cache', 'input-cache', 'arena', 'weights',
                 'model-weights'}


def sha(raw):
    return hashlib.sha256(raw).hexdigest()


def encode(value):
    return (json.dumps(value, indent=2, sort_keys=True) + '\n').encode()


def identity(s):
    return dict(device=s.st_dev, inode=s.st_ino, bytes=s.st_size,
                mtime_ns=s.st_mtime_ns, ctime_ns=s.st_ctime_ns,
                mode=stat.S_IMODE(s.st_mode))


def stable_read(path):
    before = path.lstat()
    if not stat.S_ISREG(before.st_mode):
        raise RuntimeError('not an original regular file: ' + str(path))
    fd = os.open(path, os.O_RDONLY | getattr(os, 'O_NOFOLLOW', 0))
    with os.fdopen(fd, 'rb') as stream:
        opened = os.fstat(stream.fileno())
        raw = stream.read()
        after_fd = os.fstat(stream.fileno())
    after = path.lstat()
    values = [identity(s) for s in (before, opened, after_fd, after)]
    if any(v != values[0] for v in values) or len(raw) != before.st_size:
        raise RuntimeError('closed source changed during copy: ' + str(path))
    return raw, values[0]


def write_new(relative, raw):
    target = STAGE / relative
    target.parent.mkdir(parents=True, exist_ok=True)
    with target.open('xb') as stream:
        stream.write(raw)
    if target.read_bytes() != raw:
        raise RuntimeError('destination readback mismatch: ' + relative)
    return dict(archive_path=relative, bytes=len(raw), sha256=sha(raw))


def main():
    if STAGE.exists():
        raise RuntimeError('refusing to overwrite closed-preparation packet')
    plan = json.loads(SELECTION.read_bytes())
    if set(plan['validation']) & set(plan['excluded_validation']):
        raise RuntimeError('deferred validation selected')
    if set(plan['control']) & set(plan['deferred_control'] + plan['unexecuted_prepared_controls']):
        raise RuntimeError('active or unexecuted controller selected')
    sources, excluded = [], []
    for label, root, names in [('validation', VALIDATION, plan['validation']),
                               ('control', CONTROL, plan['control'])]:
        for name in names:
            if '/' in name or name in ('.', '..'):
                raise RuntimeError('selection must use explicit top-level names')
            top = root / name
            if top.is_symlink() or not top.exists():
                raise RuntimeError('missing/symlinked selected original: ' + str(top))
            if top.is_file():
                sources.append((label, root, top))
                continue
            for base, dirs, files in os.walk(top, followlinks=False):
                base = Path(base)
                keep = []
                for name in sorted(dirs):
                    path = base / name
                    if name in EXCLUDED_DIRS or path.is_symlink():
                        excluded.append(dict(original_path=str(path), kind='directory',
                                             reason='temporary/runtime payload or symlink; contents not read'))
                    else:
                        keep.append(name)
                dirs[:] = keep
                for name in sorted(files):
                    path = base / name
                    if path.is_symlink():
                        excluded.append(dict(original_path=str(path), kind='symlink', reason='not followed'))
                    elif path.suffix.lower() in TEXT_SUFFIXES or path.name in TEXT_NAMES:
                        sources.append((label, root, path))
                    else:
                        excluded.append(dict(original_path=str(path), kind='file', bytes=path.stat().st_size,
                                             reason='payload/database/binary outside text-metadata allowlist'))
    observed_bytes = sum(path.stat().st_size for _, _, path in sources)
    if len(sources) != plan['estimated_payload_files'] or observed_bytes != plan['estimated_payload_bytes']:
        raise RuntimeError(json.dumps(dict(selection_inventory_changed=True,
            expected_files=plan['estimated_payload_files'], observed_files=len(sources),
            expected_bytes=plan['estimated_payload_bytes'], observed_bytes=observed_bytes), sort_keys=True))
    for original, expected in plan.get('expected_original_sha256', {}).items():
        raw, _ = stable_read(Path(original))
        if sha(raw) != expected:
            raise RuntimeError('pinned original identity mismatch: ' + original)
    STAGE.mkdir()
    for original, relative in [(Path(__file__), 'metadata/collector.py.txt'),
        (SELECTION, 'metadata/selection.json'), (INVENTORY, 'metadata/top-inventory.json'),
        (TERMINALS, 'metadata/terminal-inventory-summary.jsonl')]:
        raw, _ = stable_read(original)
        write_new(relative, raw)
    for item in plan.get('extra_metadata', []):
        raw, _ = stable_read(CONTROL / item['control_name'])
        write_new(item['archive_path'], raw)
    index = []
    for label, root, original in sorted(sources, key=lambda row: str(row[2])):
        relative = original.relative_to(root)
        archive = Path('archive') / label / relative
        renamed = original.suffix.lower() in RENAME_SUFFIXES or original.name == 'Cargo.lock'
        if renamed:
            archive = archive.with_name(archive.name + '.txt')
        raw, original_stat = stable_read(original)
        expected = plan.get('expected_original_sha256', {}).get(str(original))
        if expected is not None and sha(raw) != expected:
            raise RuntimeError('source changed from frozen pin: ' + str(original))
        record = write_new(str(archive), raw)
        index.append(dict(original_path=str(original), original_root=str(root),
            original_relative_path=str(relative), source_group=label, **record,
            original_stat_before_and_after=original_stat, renamed_as_plain_text=renamed,
            original_manifest_bytes_preserved=('manifest' in original.name.lower() and original.suffix.lower() == '.json'),
            readback_verified=True, controller_text_is_not_execution_evidence=(label == 'control')))
    if len({row['archive_path'] for row in index}) != len(index):
        raise RuntimeError('archive path collision')
    write_new('original-to-archive.json', encode(dict(
        schema='trnm.original-artifact-archive-paths.v1',
        scope='Original manifests and every copied artifact preserve complete bytes. Paths resolve through this outer map; original manifests are never rewritten.',
        files=index)))
    write_new('exclusions.json', encode(dict(schema='trnm.checked-build-e6-preflight-exclusions.v1',
        deferred_validation=plan['excluded_validation'], deferred_control=plan['deferred_control'],
        unexecuted_prepared_controls=plan['unexecuted_prepared_controls'],
        deferred_followup=plan['deferred_followup'],
        source_files_outside_explicit_selection_not_traversed=True, selected_subtree_exclusions=excluded)))
    errors = []
    for row in index:
        raw = (STAGE / row['archive_path']).read_bytes()
        if len(raw) != row['bytes'] or sha(raw) != row['sha256']:
            errors.append(row['archive_path'])
    verification = dict(schema='trnm.checked-build-e6-preflight-copy-verification.v1',
        verified_files=len(index), errors=errors, payload_bytes=sum(row['bytes'] for row in index),
        original_manifests_preserved=sum(row['original_manifest_bytes_preserved'] for row in index),
        native_tests_executed_by_collector=False, source_modified=False, runtime_or_production_qualification=False)
    write_new('verification.json', encode(verification))
    if errors:
        raise RuntimeError('archive readback failed')
    build = json.loads((STAGE / 'archive/validation/checked-test-profile-s3/accepted-artifact.json').read_bytes())
    start = json.loads((STAGE / 'archive/validation/checked-test-profile-s3/start.json').read_bytes())
    e6 = json.loads((STAGE / 'archive/validation/real-native-e6e7fc414690-attempt-01-wrapper/observation.json').read_bytes())
    config = json.loads((STAGE / 'archive/control/model-config-e6e7-s3.json').read_bytes())
    audit = json.loads((STAGE / 'archive/validation/model-prepare-s3-input-pin-audit.json').read_bytes())
    probe = json.loads((STAGE / 'archive/validation/model-cgroup-resource-probe-v2.json').read_bytes())
    checked_runner = (STAGE / 'archive/control/run_checked_test_profile_s3.py.txt').read_bytes()
    e6_runner = (STAGE / 'archive/control/run_model.e6-used.py.txt').read_bytes()
    earlier = json.loads((STAGE / 'metadata/earlier-input-refs.json').read_bytes())['earlier_packet_input_refs']
    source_ref = next(x for x in earlier if x['original_relative_path'] == 'source-inputs.native-v2.json')
    audit_ref = next(x for x in earlier if x['original_relative_path'] == 'source-inputs.native-v2.verification.json')
    binary_refs = []
    for label, info in [('checked_main', build['frozen_binary']), ('debug_unexecuted', build['debug_snapshot'])]:
        current_size = Path(info['frozen']).stat().st_size
        if current_size != info['bytes']:
            raise RuntimeError('closed ELF size differs from original build record')
        binary_refs.append(dict(label=label, original_path=info['frozen'], bytes=info['bytes'],
            sha256=info['sha256'], current_stat_size_matches=True, payload_copied=False,
            payload_rehashed_by_collector=False,
            hash_source='Exact original accepted-artifact.json fields; no ELF payload read by collector'))
    write_new('binary-references.json', encode(dict(schema='trnm.closed-elf-references.v1', binaries=binary_refs)))
    limits = config['original_native_limits']
    phase = dict(schema='trnm.checked-build-e6-preflight-scope.v1',
        checked_runner_post_run_sha256=sha(checked_runner),
        checked_start_controller_sha256=start['controller_sha256'],
        start_binds_reused_native_controller=start['controller_sha256']=='969e55a8c721200786ca7e913e8fc8f19487cbaacf9d80cd1a7bee3442ef8b9f',
        direct_checked_self_before_launch_hash_recorded=False,
        direct_checked_prehash_gap='NOT_RECORDED; preserved, not repaired or replaced by rerun',
        build_record_model_executed=build['model_executed'],
        qualified_build_record=build['qualified_build'],
        e6_native_started=e6['native_started'], e6_passed=e6['passed'],
        e6_error=e6['error'], e6_wrapper_matches_original_observation=sha(e6_runner)==e6['wrapper_sha256'],
        e6_native_output_exists=(VALIDATION/'real-native-e6e7fc414690-attempt-01').exists(),
        probe_passed=probe['passed'], probe_model_started=probe['model_started'],
        probe_memory_max=probe['cgroup']['memory.max'], probe_cpu_max=probe['cgroup']['cpu.max'],
        independent_e6_inputs=len(audit['inputs']), source_reference_matches_config=source_ref['sha256']==config['source_sha256'],
        original_verification_reference_matches_config=audit_ref['sha256']==config['audit_sha256'],
        original_limits_unchanged=limits==dict(rss_bytes=8589934592,seconds=300,snapshot_bytes=536870912,threads=4),
        actual_231_model_results_collected=False, current_run_model_collected=False)
    if not (phase['start_binds_reused_native_controller'] and not phase['build_record_model_executed']
        and phase['e6_native_started'] is False and phase['e6_passed'] is False
        and not phase['e6_native_output_exists'] and phase['e6_wrapper_matches_original_observation']
        and phase['probe_passed'] is True and phase['probe_model_started'] is False
        and phase['probe_memory_max']=='8589934592' and phase['probe_cpu_max']=='400000 100000'
        and phase['independent_e6_inputs']==31 and phase['source_reference_matches_config']
        and phase['original_verification_reference_matches_config'] and phase['original_limits_unchanged']):
        raise RuntimeError('closed build/e6/probe scope differs from original records')
    expected_status = {'checked': ('succeeded',0), 'e6': ('failed',1), 'probe': ('succeeded',0)}
    for kind, (state, exit_code) in expected_status.items():
        status = json.loads((STAGE/'metadata'/f'{kind}-process-status.json').read_bytes())
        read = json.loads((STAGE/'metadata'/f'{kind}-process-read.json').read_bytes())
        if not (status['state']==state and status['exit_code']==exit_code and status['output_complete']
            and read['stdout']['eof'] and read['stderr']['eof']):
            raise RuntimeError('original closed status/readback mismatch')
    write_new('closed-phase-scope-verification.json', encode(phase))
    summary = dict(schema='trnm.checked-build-e6-preflight-staging.v1', result='COPIED_AND_READBACK_VERIFIED',
        created_at_utc=dt.datetime.now(dt.timezone.utc).isoformat(), copy_only=True,
        candidate_source_written=False, repository_evidence_written=False, native_tests_executed_by_collector=False,
        actual_model_results_collected=False, checked_build_records_collected=True,
        prelaunch_refusal_records_collected=True,
        source_candidate_context=plan['source_candidate_context'],
        selected_validation_top_items=len(plan['validation']), selected_control_files=len(plan['control']),
        copied_files=len(index), copied_original_bytes=verification['payload_bytes'],
        all_original_manifests_unchanged=True, original_manifest_count=verification['original_manifests_preserved'],
        generated_inventory_is_summary_not_original_manifest=True,
        closed_phase_scope=plan['closed_preparation_scope_notes'],
        append_policy='Keep this packet immutable. Collect root-confirmed later campaigns in separate sibling packets.',
        path_mapping='original-to-archive.json', phase_scope='closed-phase-scope-verification.json', binary_references='binary-references.json')
    write_new('summary.json', encode(summary))
    rows = []
    for path in sorted(STAGE.rglob('*')):
        if path.is_file():
            raw = path.read_bytes()
            rows.append(dict(path=str(path.relative_to(STAGE)), bytes=len(raw), sha256=sha(raw)))
    manifest = dict(schema='trnm.checked-build-e6-preflight-file-manifest.v1', manifest_self_excluded=True,
        files=rows, files_excluding_manifest=len(rows), bytes_excluding_manifest=sum(row['bytes'] for row in rows))
    written = write_new('file-manifest.json', encode(manifest))
    print(json.dumps(dict(output=str(STAGE), result=summary['result'], copied_original_files=len(index),
        copied_original_bytes=summary['copied_original_bytes'], manifest=written,
        summary_sha256=sha((STAGE/'summary.json').read_bytes()), files_excluding_manifest=len(rows),
        bytes_excluding_manifest=manifest['bytes_excluding_manifest']), sort_keys=True))


if __name__ == '__main__':
    main()
