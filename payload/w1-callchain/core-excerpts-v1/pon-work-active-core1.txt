810: fn verify_with_kernel<E>(
811:     challenge: Hash,
812:     expected_task: Hash,
813:     target: Hash,
814:     bytes: &[u8],
815:     progress: &mut impl FnMut(VerificationProgress) -> Result<(), E>,
816:     kernel: VerificationKernel,
817: ) -> Result<VerifiedWork, VerificationError<E>> {
818:     let checked = checked_proof_inputs(challenge, expected_task, target, bytes, kernel)?;
819:     verify_checked_with_kernel(challenge, bytes, progress, kernel, checked)
820: }

822: // Private continuation of the native verifier. Only the current invocation's
823: // checked statement/inputs may enter: callers cannot supply a cached verdict.
824: fn verify_checked_with_kernel<E>(
825:     challenge: Hash,
826:     bytes: &[u8],
827:     progress: &mut impl FnMut(VerificationProgress) -> Result<(), E>,
828:     kernel: VerificationKernel,
829:     checked: CheckedProofInputs,
830: ) -> Result<VerifiedWork, VerificationError<E>> {
831:     let CheckedProofInputs {
832:         matrices,
833:         actual_task,
834:         ticket,
835:     } = checked;
836:     let (a, tail) = matrices.split_at(CELLS);
837:     let (b, claimed) = tail.split_at(CELLS);
838:     let claimed_bytes = &bytes[4 + 2 * CELLS * 4..4 + 3 * CELLS * 4];
839:     let trace_claim = &bytes[4 + 3 * CELLS * 4..];
840:     progress(VerificationProgress::BeforeReplay).map_err(VerificationError::Cancelled)?;
841:     let evaluated = match kernel {
842:         VerificationKernel::ScalarReference => ReplayedTranscript::Reference(
843:             evaluate_transcript_with_progress(challenge, a, b, progress)?,
844:         ),
845:         VerificationKernel::Transposed => ReplayedTranscript::Optimized(
846:             evaluate_transposed_with_progress::<false, E>(challenge, a, b, progress)?,
847:         ),
848:         VerificationKernel::LimbTranscript => ReplayedTranscript::Optimized(
849:             evaluate_transposed_with_progress::<true, E>(challenge, a, b, progress)?,
850:         ),
851:         VerificationKernel::ColumnLimbTranscript => {
852:             ReplayedTranscript::Optimized(evaluate_transcript_kernel_with_progress::<
853:                 false,
854:                 true,
855:                 false,
856:                 E,
857:             >(challenge, a, b, progress)?)
858:         }
859:         VerificationKernel::FullLimb => {
860:             ReplayedTranscript::Optimized(evaluate_transcript_kernel_with_progress::<
861:                 true,
862:                 false,
863:                 true,
864:                 E,
865:             >(challenge, a, b, progress)?)
866:         }
867:     };
868:     // PNW1 submits only a final digest: every transcript update is still replayed.
869:     // A mismatch skips product corrections; a matching digest is not yet VerifiedWork.
870:     if evaluated.trace().as_slice() != trace_claim {
871:         return Err(WorkError::Transcript.into());
872:     }
873:     let product = match evaluated {
874:         ReplayedTranscript::Reference(evaluated) => {
875:             finish_product_with_progress(a, evaluated, progress)
876:         }
877:         ReplayedTranscript::Optimized(evaluated) => {
878:             if matches!(kernel, VerificationKernel::FullLimb) {
879:                 finish_transposed_kernel_with_progress::<true, E>(a, evaluated, progress)
880:             } else {
881:                 finish_transposed_with_progress(a, evaluated, progress)
882:             }
883:         }
884:     }
885:     .map_err(VerificationError::Cancelled)?;
886:     let matches_claim = match kernel {
887:         VerificationKernel::ScalarReference => product == claimed,
888:         VerificationKernel::Transposed
889:         | VerificationKernel::LimbTranscript
890:         | VerificationKernel::FullLimb
891:         | VerificationKernel::ColumnLimbTranscript => product
892:             .iter()
893:             .flat_map(|word| word.to_le_bytes())
894:             .eq(claimed_bytes.iter().copied()),
895:     };
896:     if !matches_claim {
897:         return Err(WorkError::Product.into());
898:     }
899:     progress(VerificationProgress::BeforeVerifiedWork).map_err(VerificationError::Cancelled)?;
900:     Ok(VerifiedWork {
901:         challenge,
902:         task: actual_task,
903:         ticket,
904:         product,
905:     })
906: }

909: // one subtraction are exact. The prepared producer and retained transposed dots
910: // use this arithmetic, as do field additions in the limb paths. The separately
911: // retained scalar reference uses u128 remainder.
