42: // Match the existing receiver's checked Linux clock semantics without making a
43: // private receiver type public or adding a dependency. Missing clocks stay None.
44: struct ThreadCpuStamp {
45:     owner: thread::ThreadId,
46:     value: u64,
47: }
48: impl ThreadCpuStamp {
49:     fn start() -> Option<Self> {
50:         Some(Self {
51:             owner: thread::current().id(),
52:             value: thread_cpu_ns()?,
53:         })
54:     }
55:     fn finish(self) -> Option<u64> {
56:         if self.owner != thread::current().id() {
57:             return None;
58:         }
59:         thread_cpu_ns()?.checked_sub(self.value)
60:     }
61: }
62: fn cpu_elapsed(stamp: Option<ThreadCpuStamp>) -> Option<u64> {
63:     stamp.and_then(ThreadCpuStamp::finish)
64: }
65: fn checked_cpu_ns(seconds: i64, nanoseconds: i64) -> Option<u64> {
66:     let seconds = u64::try_from(seconds).ok()?;
67:     let nanoseconds = u64::try_from(nanoseconds).ok()?;
68:     if nanoseconds >= 1_000_000_000 {
69:         return None;
70:     }
71:     seconds.checked_mul(1_000_000_000)?.checked_add(nanoseconds)
72: }

75:     use rustix::time::{clock_gettime_dynamic, ClockId, DynamicClockId};
76:     let value = clock_gettime_dynamic(DynamicClockId::Known(ClockId::ThreadCPUTime)).ok()?;
77:     checked_cpu_ns(value.tv_sec, value.tv_nsec)
78: }
