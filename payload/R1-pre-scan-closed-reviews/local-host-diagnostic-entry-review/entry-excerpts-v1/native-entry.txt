76: def read_format_file(path: Path, limit: int) -> bytes:
77:     """Own one bounded regular-file read; a pathname change cannot change these bytes."""
78:     with os.fdopen(os.open(path, os.O_RDONLY | os.O_NOFOLLOW | os.O_NONBLOCK), 'rb') as stream:
79:         info = os.fstat(stream.fileno())
80:         require(stat.S_ISREG(info.st_mode) and info.st_nlink == 1, 'FORMAT_REGULAR_FILE_REQUIRED')
81:         raw = stream.read(limit + 1)
82:     require(len(raw) <= limit, 'FORMAT_RECEIPT_BOUND')
83:     return raw

135:                      'scope': 'exact-source-preparation-outside-service'}

145: def select_binary(lines: list[str]) -> Path:
146:     paths = []
147:     for line in lines:

163: 
164:     Retain both the shared support and test-local modules; flattening the test

168:     """

173:     support.mkdir(parents=True, exist_ok=True)
