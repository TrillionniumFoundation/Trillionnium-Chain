// One root-provisioned kernel parent checker shared by the V3 Rust library
// and CLI. The CLI's network-namespace slots count processes locally but are
// NOT host-wide CPU limits: Linux network namespaces can be multiplied.
// A cgroup-v2 parent is an actual aggregate kernel CPU, memory and pids cap
// over all descendant services, including different network namespaces.
//
// This code is an unprivileged reader: it cannot create, replace, widen,
// delegate, remount or migrate an OS resource group. Only an administrator can
// provision/start the fixed parent and place the process inside it. The
// original continuous ledger still owns durable operation reconciliation.
/// Actual kernel counters for the *whole original parent*, not a guest,
/// a given V3 listener or the volatile CPU-credit epoch. Resource samples
/// must never grant W1/transaction/model authority.
#[derive(Clone, Copy, Debug, PartialEq, Eq, serde::Serialize)]
pub struct KernelResourceSample {
    pub cpu_usage_usec: u64,
    pub cpu_nr_periods: u64,
    pub cpu_nr_throttled: u64,
    pub cpu_throttled_usec: u64,
    pub memory_current_bytes: u64,
    pub pids_current: u64,
}

/// These are cumulative kernel CPU deltas; the current memory/task values
/// belong to independent point-in-time samples, not to a delta.
#[derive(Clone, Copy, Debug, PartialEq, Eq, serde::Serialize)]
pub struct KernelCpuDelta {
    pub usage_usec: u64,
    pub nr_periods: u64,
    pub nr_throttled: u64,
    pub throttled_usec: u64,
}

/// The exact immutable values pinned by the live kernel-parent handle. This is
/// an observation binding, not a signed proof or authority to widen a limit.
/// The parent covers all descendant processes/threads, not only mutation work.
#[derive(Clone, Debug, PartialEq, Eq, serde::Serialize)]
pub struct HostKernelBinding {
    pub schema: &'static str,
    pub parent: String,
    pub boot_id: String,
    pub device: u64,
    pub inode: u64,
    pub cpu_quota_usec: u64,
    pub cpu_period_usec: u64,
    pub cpu_burst_usec: u64,
    pub memory_max_bytes: u64,
    pub memory_swap_max_bytes: u64,
    pub pids_max: u64,
}

/// Counters from the same verified fixed parent, with its original identity and
/// limits attached for an independent reader. Memory/tasks remain point samples.
/// A manually constructed diagnostic value conveys no OS or ledger authority.
#[derive(Clone, Debug, PartialEq, Eq, serde::Serialize)]
pub struct BoundKernelResourceSample {
    pub schema: &'static str,
    pub binding: HostKernelBinding,
    pub resources: KernelResourceSample,
}
impl BoundKernelResourceSample {
    /// A different boot, inode, fixed parent or any changed resource ceiling
    /// cannot be used as the other endpoint of the same accounting interval.
    pub fn cpu_delta_since(&self, before: &Self) -> crate::Result<KernelCpuDelta> {
        crate::ensure(
            self.schema == "public-host-kernel-bound-resource-sample-v1"
                && before.schema == self.schema
                && self.binding.schema == "public-host-kernel-binding-v1"
                && self.binding == before.binding,
            "PUBLIC_HOST_KERNEL_BINDING_CHANGED",
        )?;
        self.resources.cpu_delta_since(before.resources)
    }
}

impl KernelResourceSample {
    pub fn cpu_delta_since(self, before: Self) -> crate::Result<KernelCpuDelta> {
        let changed = || crate::Error::from("PUBLIC_HOST_KERNEL_COUNTER_REWIND");
        Ok(KernelCpuDelta {
            usage_usec: self
                .cpu_usage_usec
                .checked_sub(before.cpu_usage_usec)
                .ok_or_else(changed)?,
            nr_periods: self
                .cpu_nr_periods
                .checked_sub(before.cpu_nr_periods)
                .ok_or_else(changed)?,
            nr_throttled: self
                .cpu_nr_throttled
                .checked_sub(before.cpu_nr_throttled)
                .ok_or_else(changed)?,
            throttled_usec: self
                .cpu_throttled_usec
                .checked_sub(before.cpu_throttled_usec)
                .ok_or_else(changed)?,
        })
    }
}

#[cfg(target_os = "linux")]
mod kernel_parent {
    use super::{BoundKernelResourceSample, HostKernelBinding, KernelResourceSample};
    use std::{
        fs,
        io::{self, Read},
        net::SocketAddr,
        os::unix::fs::{MetadataExt, PermissionsExt},
        path::{Path, PathBuf},
    };

    type Result<T> = crate::Result<T>;
    const CGROUP_ROOT: &str = "/sys/fs/cgroup";
    // systemd.slice(5): a no-dash slice name belongs directly to -.slice.
    // system.slice children must instead be named system-<child>.slice.
    // Keep the one original fixed name; do NOT admit caller-selectable slices.
    const PARENT_DIRECTORY: &str = "/sys/fs/cgroup";
    // This is ONE administrator-provisioned parent for all public V3
    // processes. Allowing caller-chosen trnmpon*.slice names would let
    // multiple independent 25%-CPU parents multiply the physical budget.
    const PARENT_NAME: &str = "trnmponpublic.slice";
    fn approved_parent_path(path: &Path) -> bool {
        path.parent() == Some(Path::new(PARENT_DIRECTORY))
            && path.file_name().and_then(|part| part.to_str()) == Some(PARENT_NAME)
    }
    const MAX_MEMORY_BYTES: u64 = 2 * 1024 * 1024 * 1024;
    const MAX_TASKS: u64 = 256;
    const LIMIT_NAMES: [&str; 5] = [
        "cpu.max",
        "cpu.max.burst",
        "memory.max",
        "memory.swap.max",
        "pids.max",
    ];
    #[derive(Debug, Clone, Copy, PartialEq, Eq)]
    struct Limits {
        cpu_quota_us: u64,
        cpu_period_us: u64,
        memory_bytes: u64,
        tasks: u64,
    }

    // Open the current path on EVERY invocation, with current credentials.
    // No descriptor, content, permission or resource verdict is retained.
    // Small procfs/kernfs values do not need read_to_string's size probing and
    // growth loop. Still read to a real EOF, including after a short read;
    // Interrupted is retried and every other read failure stays a failure.
    // Only the existing quota recheck has a 128-byte ceiling. Boot identity,
    // thread membership and initial limit parsing retain their original size
    // behavior instead of imposing a new successful-input restriction.
    fn read_fresh_kernel_text(
        path: impl AsRef<Path>,
        maximum: Option<usize>,
    ) -> io::Result<String> {
        read_kernel_text(&mut fs::File::open(path)?, maximum)
    }

    fn read_kernel_text(reader: &mut impl Read, maximum: Option<usize>) -> io::Result<String> {
        let limit = maximum
            .map(|maximum| {
                maximum
                    .checked_add(1)
                    .ok_or_else(|| io::Error::new(io::ErrorKind::InvalidData, "kernel text limit"))
            })
            .transpose()?
            .unwrap_or(usize::MAX);
        let mut bytes = Vec::new();
        let mut buffer = [0_u8; 4096];
        loop {
            let available = (limit - bytes.len()).min(buffer.len());
            let count = match reader.read(&mut buffer[..available]) {
                Ok(count) => count,
                Err(error) if error.kind() == io::ErrorKind::Interrupted => continue,
                Err(error) => return Err(error),
            };
            if count == 0 {
                return String::from_utf8(bytes)
                    .map_err(|error| io::Error::new(io::ErrorKind::InvalidData, error));
            }
            bytes.extend_from_slice(&buffer[..count]);
            if maximum.is_some_and(|maximum| bytes.len() > maximum) {
                return Err(io::Error::new(
                    io::ErrorKind::InvalidData,
                    "kernel text limit",
                ));
            }
        }
    }

    /// Ephemeral text for the five quota rechecks that already have a 128-byte
    /// ceiling. This owns no descriptor or previous observation. The extra byte
    /// distinguishes an exact-limit value followed by real EOF from oversize
    /// input; a late read error is still an error rather than a valid prefix.
    struct BoundedKernelText {
        bytes: [u8; 129],
        length: usize,
    }

    impl std::ops::Deref for BoundedKernelText {
        type Target = str;

        fn deref(&self) -> &str {
            // Only read_bounded_kernel_text constructs this private value and
            // checks the complete UTF-8 input before exposing it.
            std::str::from_utf8(&self.bytes[..self.length]).expect("validated bounded kernel text")
        }
    }

    fn read_bounded_kernel_text(reader: &mut impl Read) -> io::Result<BoundedKernelText> {
        let mut text = BoundedKernelText {
            bytes: [0; 129],
            length: 0,
        };
        loop {
            let count = match reader.read(&mut text.bytes[text.length..]) {
                Ok(count) => count,
                Err(error) if error.kind() == io::ErrorKind::Interrupted => continue,
                Err(error) => return Err(error),
            };
            if count == 0 {
                std::str::from_utf8(&text.bytes[..text.length])
                    .map_err(|error| io::Error::new(io::ErrorKind::InvalidData, error))?;
                return Ok(text);
            }
            text.length += count;
            if text.length > 128 {
                return Err(io::Error::new(
                    io::ErrorKind::InvalidData,
                    "kernel text limit",
                ));
            }
        }
    }

    fn read_current_bounded_limits(paths: &[PathBuf; 5]) -> Result<Limits> {
        let read = |index: usize| -> Result<BoundedKernelText> {
            // Reopen the current path with the current credentials on EVERY
            // check. Only immutable path concatenation is prepared in open().
            let mut file = fs::File::open(&paths[index])
                .map_err(|_| crate::Error::from("PUBLIC_HOST_KERNEL_UNAVAILABLE"))?;
            read_bounded_kernel_text(&mut file)
                .map_err(|_| crate::Error::from("PUBLIC_HOST_KERNEL_UNAVAILABLE"))
        };
        // Preserve read order and error precedence: all five original reads
        // must succeed before any limit is parsed or compared.
        validate_limits(&read(0)?, &read(1)?, &read(2)?, &read(3)?, &read(4)?)
    }

    fn numeric(text: &str, limit: u64) -> Option<u64> {
        let value = text.trim().parse::<u64>().ok()?;
        (value > 0 && value <= limit).then_some(value)
    }

    // The maximum CPU bandwidth is strictly <= 250,000,000 ns per second per
    // entire host parent. Explicit finite ceilings only; never accept "max".
    fn validate_limits(
        cpu: &str,
        burst: &str,
        memory: &str,
        swap: &str,
        pids: &str,
    ) -> Result<Limits> {
        let mut parts = cpu.split_ascii_whitespace();
        let Some((quota, period)) = parts.next().zip(parts.next()) else {
            return Err("PUBLIC_HOST_KERNEL_LIMITS".into());
        };
        if parts.next().is_some() {
            return Err("PUBLIC_HOST_KERNEL_LIMITS".into());
        }
        let quota = numeric(quota, 250_000).ok_or("PUBLIC_HOST_KERNEL_LIMITS")?;
        let period = numeric(period, 1_000_000).ok_or("PUBLIC_HOST_KERNEL_LIMITS")?;
        let memory = numeric(memory, MAX_MEMORY_BYTES).ok_or("PUBLIC_HOST_KERNEL_LIMITS")?;
        let tasks = numeric(pids, MAX_TASKS).ok_or("PUBLIC_HOST_KERNEL_LIMITS")?;
        if quota.checked_mul(4).is_none_or(|value| value > period)
            || burst.trim() != "0"
            || swap.trim() != "0"
        {
            return Err("PUBLIC_HOST_KERNEL_LIMITS".into());
        }
        Ok(Limits {
            cpu_quota_us: quota,
            cpu_period_us: period,
            memory_bytes: memory,
            tasks,
        })
    }

    fn own_cgroup() -> Result<String> {
        // A process may have independently placed native worker threads in a
        // cgroup-v2 threaded subtree. /proc/self identifies the thread-group
        // leader and cannot attest the actual calling worker's CPU cgroup.
        // The kernel's /proc/thread-self resolves to the current TID.
        let text = read_fresh_kernel_text("/proc/thread-self/cgroup", None)
            .map_err(|_| crate::Error::from("PUBLIC_HOST_KERNEL_CONTEXT"))?;
        let mut entries = text.lines();
        let only = entries.next().ok_or("PUBLIC_HOST_KERNEL_CONTEXT")?;
        if entries.next().is_some() {
            return Err("PUBLIC_HOST_KERNEL_CONTEXT".into());
        }
        let relative = only
            .strip_prefix("0::/")
            .ok_or("PUBLIC_HOST_KERNEL_CONTEXT")?;
        if relative.is_empty()
            || relative
                .split('/')
                .any(|x| x.is_empty() || x == "." || x == "..")
        {
            return Err("PUBLIC_HOST_KERNEL_CONTEXT".into());
        }
        Ok(relative.to_owned())
    }

    /// Accept additional future kernel cpu.stat keys, but never fabricate the
    /// four required counters, accept duplicates or accept malformed/huge input.
    /// The actual sysfs files are read only after the fixed parent was checked.
    fn resource_sample(cpu: &str, memory: &str, tasks: &str) -> Result<KernelResourceSample> {
        const INVALID: &str = "PUBLIC_HOST_KERNEL_COUNTERS";
        if cpu.len() > 2048 || memory.len() > 64 || tasks.len() > 64 {
            return Err(INVALID.into());
        }
        let mut fields = std::collections::BTreeMap::<&str, u64>::new();
        for line in cpu.lines() {
            let mut words = line.split_ascii_whitespace();
            let key = words.next().ok_or(INVALID)?;
            let raw = words.next().ok_or(INVALID)?;
            if words.next().is_some() {
                return Err(INVALID.into());
            }
            let value = raw
                .parse::<u64>()
                .map_err(|_| crate::Error::from(INVALID))?;
            if fields.insert(key, value).is_some() {
                return Err(INVALID.into());
            }
        }
        let required = |key| {
            fields
                .get(key)
                .copied()
                .ok_or_else(|| crate::Error::from(INVALID))
        };
        let current = |raw: &str| -> Result<u64> {
            let mut parts = raw.split_ascii_whitespace();
            let value = parts
                .next()
                .ok_or(INVALID)?
                .parse::<u64>()
                .map_err(|_| crate::Error::from(INVALID))?;
            if parts.next().is_some() {
                return Err(INVALID.into());
            }
            Ok(value)
        };
        Ok(KernelResourceSample {
            cpu_usage_usec: required("usage_usec")?,
            cpu_nr_periods: required("nr_periods")?,
            cpu_nr_throttled: required("nr_throttled")?,
            cpu_throttled_usec: required("throttled_usec")?,
            memory_current_bytes: current(memory)?,
            pids_current: current(tasks)?,
        })
    }

    pub struct KernelBudget {
        parent: PathBuf,
        // Immutable derived path bytes, never open descriptors, quota contents,
        // permissions, membership or a successful resource-check verdict.
        limit_paths: [PathBuf; 5],
        identity: (u64, u64),
        boot_id: String,
        limits: Limits,
    }
    impl KernelBudget {
        fn check(&self) -> Result<()> {
            let now_boot = read_fresh_kernel_text("/proc/sys/kernel/random/boot_id", None)
                .map_err(|_| crate::Error::from("PUBLIC_HOST_KERNEL_CONTEXT"))?;
            if now_boot.trim() != self.boot_id {
                return Err("PUBLIC_HOST_KERNEL_REPLACED".into());
            }
            let meta = fs::symlink_metadata(&self.parent)
                .map_err(|_| crate::Error::from("PUBLIC_HOST_KERNEL_REPLACED"))?;
            if !meta.file_type().is_dir()
                || meta.uid() != 0
                || meta.mode() & 0o022 != 0
                || (meta.dev(), meta.ino()) != self.identity
            {
                return Err("PUBLIC_HOST_KERNEL_REPLACED".into());
            }
            let parent_rel = self
                .parent
                .strip_prefix(CGROUP_ROOT)
                .map_err(|_| crate::Error::from("PUBLIC_HOST_KERNEL_CONTEXT"))?
                .to_str()
                .ok_or("PUBLIC_HOST_KERNEL_CONTEXT")?
                .trim_start_matches('/');
            let current = own_cgroup()?;
            if current != parent_rel
                && !current
                    .strip_prefix(parent_rel)
                    .is_some_and(|suffix| suffix.starts_with('/'))
            {
                return Err("PUBLIC_HOST_KERNEL_CONTEXT".into());
            }
            let observed = read_current_bounded_limits(&self.limit_paths)?;
            if observed != self.limits {
                return Err("PUBLIC_HOST_KERNEL_LIMITS_CHANGED".into());
            }
            Ok(())
        }
        fn open(path: &str) -> Result<Self> {
            let user_path = Path::new(path);
            if !user_path.is_absolute() {
                return Err("PUBLIC_HOST_KERNEL_PARENT".into());
            }
            let parent = fs::canonicalize(user_path)
                .map_err(|_| crate::Error::from("PUBLIC_HOST_KERNEL_PARENT"))?;
            // A strict systemd-owned *single parent*, not the common cgroup
            // root, any child/service, or a caller-created user.slice.
            if !approved_parent_path(&parent) || parent != user_path {
                return Err("PUBLIC_HOST_KERNEL_PARENT".into());
            }
            if !Path::new("/sys/fs/cgroup/cgroup.controllers").is_file() {
                return Err("PUBLIC_HOST_KERNEL_UNSUPPORTED".into());
            }
            let meta = fs::symlink_metadata(&parent)?;
            if !meta.is_dir() || meta.uid() != 0 || meta.permissions().mode() & 0o022 != 0 {
                return Err("PUBLIC_HOST_KERNEL_PARENT".into());
            }
            let boot = read_fresh_kernel_text("/proc/sys/kernel/random/boot_id", None)
                .map_err(|_| crate::Error::from("PUBLIC_HOST_KERNEL_CONTEXT"))?;
            let budget = Self {
                identity: (meta.dev(), meta.ino()),
                boot_id: boot.trim().to_owned(),
                limit_paths: LIMIT_NAMES.map(|name| parent.join(name)),
                parent,
                limits: Limits {
                    cpu_quota_us: 0,
                    cpu_period_us: 0,
                    memory_bytes: 0,
                    tasks: 0,
                },
            };
            let read = |index: usize| -> Result<String> {
                // Initial parsing retains its original unbounded text grammar.
                read_fresh_kernel_text(&budget.limit_paths[index], None)
                    .map_err(|_| crate::Error::from("PUBLIC_HOST_KERNEL_UNAVAILABLE"))
            };
            let limits = validate_limits(&read(0)?, &read(1)?, &read(2)?, &read(3)?, &read(4)?)?;
            let checked = Self { limits, ..budget };
            checked.check()?;
            Ok(checked)
        }
        pub fn verify(&self) -> Result<()> {
            self.check()
        }

        /// Source-independent identity observation for one explicit local
        /// resource-policy epoch. It is never a reusable successful check.
        pub(crate) fn identity_observation(&self) -> Result<serde_json::Value> {
            self.check()?;
            let value = serde_json::json!({
                "schema":"fixed-kernel-parent-identity-observation-v1",
                "path":self.parent,"device":self.identity.0,"inode":self.identity.1,
                "boot_id":self.boot_id,"cpu_quota_us":self.limits.cpu_quota_us,
                "cpu_period_us":self.limits.cpu_period_us,"cpu_burst_us":0,
                "memory_bytes":self.limits.memory_bytes,"swap_bytes":0,"tasks":self.limits.tasks,
                "observation_is_authority":false,
            });
            self.check()?;
            Ok(value)
        }

        /// Read-only aggregate kernel accounting. Verify the same fixed
        /// boot/inode/quotas/membership both before and after sampling; if any
        /// counter is unavailable or malformed, never issue a qualified receipt.
        pub fn resource_sample(&self) -> Result<KernelResourceSample> {
            self.check()?;
            let read = |name| -> Result<String> {
                read_fresh_kernel_text(self.parent.join(name), None)
                    .map_err(|_| crate::Error::from("PUBLIC_HOST_KERNEL_COUNTERS"))
            };
            let sample = resource_sample(
                &read("cpu.stat")?,
                &read("memory.current")?,
                &read("pids.current")?,
            )?;
            self.check()?;
            Ok(sample)
        }

        /// Uses the same checked sample path: fixed parent, boot/inode, actual
        /// calling-thread membership and exact unchanged limits before and after
        /// all reads. It creates no cgroup, reservation, credit or policy change.
        pub fn bound_resource_sample(&self) -> Result<BoundKernelResourceSample> {
            let resources = self.resource_sample()?;
            Ok(BoundKernelResourceSample {
                schema: "public-host-kernel-bound-resource-sample-v1",
                binding: HostKernelBinding {
                    schema: "public-host-kernel-binding-v1",
                    parent: self
                        .parent
                        .to_str()
                        .ok_or("PUBLIC_HOST_KERNEL_PARENT")?
                        .to_owned(),
                    boot_id: self.boot_id.clone(),
                    device: self.identity.0,
                    inode: self.identity.1,
                    cpu_quota_usec: self.limits.cpu_quota_us,
                    cpu_period_usec: self.limits.cpu_period_us,
                    // Both exact zero settings are revalidated on every sample.
                    cpu_burst_usec: 0,
                    memory_max_bytes: self.limits.memory_bytes,
                    memory_swap_max_bytes: 0,
                    pids_max: self.limits.tasks,
                },
                resources,
            })
        }
    }

    pub fn require(address: SocketAddr, parent: Option<&str>) -> Result<Option<KernelBudget>> {
        match parent {
            Some(path) => KernelBudget::open(path).map(Some),
            None if !address.ip().is_loopback() => Err("PUBLIC_HOST_KERNEL_REQUIRED".into()),
            None => Ok(None), // Retained loopback-only development tools.
        }
    }

    #[cfg(test)]
    mod fresh_read_tests {
        use super::*;

        struct ShortReader<'a> {
            remaining: &'a [u8],
            consumed: usize,
            interrupt: bool,
            tail_error: Option<io::ErrorKind>,
        }
        impl Read for ShortReader<'_> {
            fn read(&mut self, destination: &mut [u8]) -> io::Result<usize> {
                if self.interrupt {
                    self.interrupt = false;
                    return Err(io::ErrorKind::Interrupted.into());
                }
                if self.remaining.is_empty() {
                    return self.tail_error.map_or(Ok(0), |kind| Err(kind.into()));
                }
                let count = destination.len().min(self.remaining.len()).min(3);
                destination[..count].copy_from_slice(&self.remaining[..count]);
                self.remaining = &self.remaining[count..];
                self.consumed += count;
                Ok(count)
            }
        }

        #[test]
        fn exact_limit_requires_real_eof_and_overlimit_read_stays_bounded() {
            for length in [0, 1, 127, 128, 129, 16_384] {
                let value = vec![b'7'; length];
                let mut reader = ShortReader {
                    remaining: &value,
                    consumed: 0,
                    interrupt: false,
                    tail_error: None,
                };
                let result = read_kernel_text(&mut reader, Some(128));
                if length <= 128 {
                    assert_eq!(result.unwrap().as_bytes(), value);
                    assert_eq!(reader.consumed, length);
                } else {
                    assert_eq!(result.unwrap_err().kind(), io::ErrorKind::InvalidData);
                    assert_eq!(reader.consumed, 129);
                }
            }
            let mut exact_with_late_error = ShortReader {
                remaining: &[b'7'; 128],
                consumed: 0,
                interrupt: false,
                tail_error: Some(io::ErrorKind::PermissionDenied),
            };
            assert_eq!(
                read_kernel_text(&mut exact_with_late_error, Some(128))
                    .unwrap_err()
                    .kind(),
                io::ErrorKind::PermissionDenied
            );
        }

        #[test]
        fn short_reads_and_interruption_preserve_unbounded_original_text() {
            let value = "0::/trnmponpublic.slice/".to_owned() + &"a".repeat(9000) + ".service\n";
            let mut reader = ShortReader {
                remaining: value.as_bytes(),
                consumed: 0,
                interrupt: true,
                tail_error: None,
            };
            assert_eq!(read_kernel_text(&mut reader, None).unwrap(), value);
            assert_eq!(reader.consumed, value.len());
        }

        #[test]
        fn invalid_utf8_and_io_errors_never_become_a_successful_prefix() {
            for maximum in [None, Some(128)] {
                let mut invalid = &b"25000 100000\n\xff"[..];
                assert_eq!(
                    read_kernel_text(&mut invalid, maximum).unwrap_err().kind(),
                    io::ErrorKind::InvalidData
                );
                for prefix in [&b""[..], &b"25000 100000\n"[..]] {
                    let mut failed = ShortReader {
                        remaining: prefix,
                        consumed: 0,
                        interrupt: false,
                        tail_error: Some(io::ErrorKind::PermissionDenied),
                    };
                    assert_eq!(
                        read_kernel_text(&mut failed, maximum).unwrap_err().kind(),
                        io::ErrorKind::PermissionDenied
                    );
                }
            }
        }

        fn original_limits(
            cpu: &str,
            burst: &str,
            memory: &str,
            swap: &str,
            pids: &str,
        ) -> Result<Limits> {
            // Retain the pre-change allocation-based parser as an independent
            // reference, including its token-count and numeric error order.
            let parts: Vec<_> = cpu.split_ascii_whitespace().collect();
            if parts.len() != 2 {
                return Err("PUBLIC_HOST_KERNEL_LIMITS".into());
            }
            let quota = numeric(parts[0], 250_000).ok_or("PUBLIC_HOST_KERNEL_LIMITS")?;
            let period = numeric(parts[1], 1_000_000).ok_or("PUBLIC_HOST_KERNEL_LIMITS")?;
            let memory = numeric(memory, MAX_MEMORY_BYTES).ok_or("PUBLIC_HOST_KERNEL_LIMITS")?;
            let tasks = numeric(pids, MAX_TASKS).ok_or("PUBLIC_HOST_KERNEL_LIMITS")?;
            if quota.checked_mul(4).is_none_or(|value| value > period)
                || burst.trim() != "0"
                || swap.trim() != "0"
            {
                return Err("PUBLIC_HOST_KERNEL_LIMITS".into());
            }
            Ok(Limits {
                cpu_quota_us: quota,
                cpu_period_us: period,
                memory_bytes: memory,
                tasks,
            })
        }

        fn original_current_limits(parent: &Path) -> Result<Limits> {
            let read = |name| -> Result<String> {
                let text = read_fresh_kernel_text(parent.join(name), Some(128))
                    .map_err(|_| crate::Error::from("PUBLIC_HOST_KERNEL_UNAVAILABLE"))?;
                if text.len() > 128 {
                    return Err("PUBLIC_HOST_KERNEL_UNAVAILABLE".into());
                }
                Ok(text)
            };
            original_limits(
                &read("cpu.max")?,
                &read("cpu.max.burst")?,
                &read("memory.max")?,
                &read("memory.swap.max")?,
                &read("pids.max")?,
            )
        }

        fn stack_quota_reads_preserve_fresh_paths_and_read_before_parse_errors() {
            let directory = tempfile::tempdir().unwrap();
            let paths = LIMIT_NAMES.map(|name| directory.path().join(name));
            let values = ["25000 100000\n", "0\n", "268435456\n", "0\n", "32\n"];
            let restore = || {
                for (path, value) in paths.iter().zip(values) {
                    fs::write(path, value).unwrap();
                }
            };
            let compare = || {
                let expected =
                    original_current_limits(directory.path()).map_err(|error| error.to_string());
                let observed =
                    read_current_bounded_limits(&paths).map_err(|error| error.to_string());
                assert_eq!(observed, expected);
                observed
            };
            restore();
            assert!(compare().is_ok());
            // A malformed first value cannot mask an unavailable later file:
            // the original implementation completes all reads before parsing.
            fs::write(&paths[0], "max 100000\n").unwrap();
            fs::remove_file(&paths[4]).unwrap();
            assert_eq!(compare().unwrap_err(), "PUBLIC_HOST_KERNEL_UNAVAILABLE");
            restore();
            for index in 0..paths.len() {
                for bytes in [vec![b'7'; 129], vec![0xff], b"bad\n".to_vec()] {
                    fs::write(&paths[index], &bytes).unwrap();
                    assert!(compare().is_err());
                    restore();
                }
                // Derived path bytes cannot retain an old file object or its
                // value across rename/replacement, removal or changed access.
                let retired = directory.path().join("retired");
                fs::rename(&paths[index], &retired).unwrap();
                fs::write(&paths[index], "1\n").unwrap();
                compare().ok();
                fs::remove_file(&paths[index]).unwrap();
                assert_eq!(compare().unwrap_err(), "PUBLIC_HOST_KERNEL_UNAVAILABLE");
                fs::rename(&retired, &paths[index]).unwrap();
                fs::set_permissions(&paths[index], fs::Permissions::from_mode(0o0)).unwrap();
                // Actual current credentials may retain DAC override. Both
                // implementations must observe the same real kernel outcome.
                compare().ok();
                fs::set_permissions(&paths[index], fs::Permissions::from_mode(0o600)).unwrap();
                restore();
            }
            assert!(compare().is_ok());
        }

/// Single protocol-neutral, administrator-provisioned host resource parent.
/// The public V3 library and development CLI use the EXACT same eligibility.
pub const FIXED_PUBLIC_KERNEL_PARENT: &str = "/sys/fs/cgroup/trnmponpublic.slice";
/// Direct library callers cannot bypass the CLI's kernel-parent requirement.
/// A loopback test service needs no administrator privileges; an externally
/// reachable service must already be placed under the one fixed parent.
pub fn require_for_listener(
    address: std::net::SocketAddr,
) -> crate::Result<Option<HostKernelBudget>> {
    require_host_kernel_budget(
        address,
        (!address.ip().is_loopback()).then_some(FIXED_PUBLIC_KERNEL_PARENT),
    )
}

                10 => after.schema = "different-sample-schema",
                _ => unreachable!(),
            }
            assert_eq!(
                after.cpu_delta_since(&before).unwrap_err().to_string(),
                "PUBLIC_HOST_KERNEL_BINDING_CHANGED",
                "change {change}"
            );
        }
    }

    #[test]
    fn unchanged_parent_identity_does_not_excuse_a_kernel_cpu_counter_rewind() {
        let before = sample();
        for change in 0..4 {
            let mut after = before.clone();
            match change {
                0 => after.resources.cpu_usage_usec -= 1,
                1 => after.resources.cpu_nr_periods -= 1,
                2 => after.resources.cpu_nr_throttled -= 1,
                3 => after.resources.cpu_throttled_usec -= 1,
                _ => unreachable!(),
            }
            assert_eq!(
                after.cpu_delta_since(&before).unwrap_err().to_string(),
                "PUBLIC_HOST_KERNEL_COUNTER_REWIND",
                "counter {change}"
            );
        }
    }
