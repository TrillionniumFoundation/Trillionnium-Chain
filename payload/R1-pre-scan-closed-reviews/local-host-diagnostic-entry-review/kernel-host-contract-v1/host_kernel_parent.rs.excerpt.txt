1: // One root-provisioned kernel parent checker shared by the V3 Rust library
2: // and CLI. The CLI's network-namespace slots count processes locally but are
3: // NOT host-wide CPU limits: Linux network namespaces can be multiplied.
4: // A cgroup-v2 parent is an actual aggregate kernel CPU, memory and pids cap
5: // over all descendant services, including different network namespaces.
6: //
7: // This code is an unprivileged reader: it cannot create, replace, widen,
8: // delegate, remount or migrate an OS resource group. Only an administrator can
9: // provision/start the fixed parent and place the process inside it. The
10: // original continuous ledger still owns durable operation reconciliation.
11: /// Actual kernel counters for the *whole original parent*, not a guest,
12: /// a given V3 listener or the volatile CPU-credit epoch. Resource samples
13: /// must never grant W1/transaction/model authority.
14: #[derive(Clone, Copy, Debug, PartialEq, Eq, serde::Serialize)]
15: pub struct KernelResourceSample {
16:     pub cpu_usage_usec: u64,
17:     pub cpu_nr_periods: u64,
18:     pub cpu_nr_throttled: u64,
19:     pub cpu_throttled_usec: u64,
20:     pub memory_current_bytes: u64,
21:     pub pids_current: u64,
22: }
23:

25: /// belong to independent point-in-time samples, not to a delta.

27: pub struct KernelCpuDelta {
28:     pub usage_usec: u64,
29:     pub nr_periods: u64,
30:     pub nr_throttled: u64,

35:     pub fn cpu_delta_since(self, before: Self) -> crate::Result<KernelCpuDelta> {
36:         let changed = || crate::Error::from("PUBLIC_HOST_KERNEL_COUNTER_REWIND");
37:         Ok(KernelCpuDelta {
38:             usage_usec: self
39:                 .cpu_usage_usec
40:                 .checked_sub(before.cpu_usage_usec)
41:                 .ok_or_else(changed)?,
42:             nr_periods: self
43:                 .cpu_nr_periods
44:                 .checked_sub(before.cpu_nr_periods)
45:                 .ok_or_else(changed)?,
46:             nr_throttled: self
47:                 .cpu_nr_throttled
48:                 .checked_sub(before.cpu_nr_throttled)
49:                 .ok_or_else(changed)?,
50:             throttled_usec: self
51:                 .cpu_throttled_usec
52:                 .checked_sub(before.cpu_throttled_usec)
53:                 .ok_or_else(changed)?,
54:         })
55:     }
56: }

100:     // The maximum CPU bandwidth is <= 250,000,000 ns per second per
101:     // entire host parent. Explicit finite ceilings only; never accept "max".
102:     fn validate_limits(
103:         cpu: &str,
104:         burst: &str,
105:         memory: &str,
106:         swap: &str,
107:         pids: &str,
108:     ) -> Result<Limits> {
109:         let mut parts = cpu.split_ascii_whitespace();
110:         let (quota, period) = match (parts.next(), parts.next(), parts.next()) {
111:             (Some(quota), Some(period), None) => (quota, period),
112:             _ => return Err("PUBLIC_HOST_KERNEL_LIMITS".into()),
113:         };
114:         let quota = numeric(quota, 250_000).ok_or("PUBLIC_HOST_KERNEL_LIMITS")?;
115:         let period = numeric(period, 1_000_000).ok_or("PUBLIC_HOST_KERNEL_LIMITS")?;
116:         let memory = numeric(memory, MAX_MEMORY_BYTES).ok_or("PUBLIC_HOST_KERNEL_LIMITS")?;
117:         let tasks = numeric(pids, MAX_TASKS).ok_or("PUBLIC_HOST_KERNEL_LIMITS")?;
118:         if quota.checked_mul(4).is_none_or(|value| value > period)
119:             || burst.trim() != "0"
120:             || swap.trim() != "0"
121:         {
122:             return Err("PUBLIC_HOST_KERNEL_LIMITS".into());
123:         }
124:         Ok(Limits {
125:             cpu_quota_us: quota,
126:             cpu_period_us: period,
127:             memory_bytes: memory,
128:             tasks,
129:         })
130:     }

136:     fn read_counter_text<'a>(reader: &mut impl Read, bytes: &'a mut [u8]) -> Result<&'a str> {
137:         // The final byte is an overflow probe. A short read is never EOF,
138:         // including a read exactly at the existing parser's byte bound.
139:         if bytes.is_empty() {
140:             return Err("PUBLIC_HOST_KERNEL_COUNTERS".into());
141:         }
142:         let mut used = 0;
143:         loop {
144:             let count = match reader.read(&mut bytes[used..]) {
145:                 Ok(count) => count,
146:                 Err(error) if error.kind() == io::ErrorKind::Interrupted => continue,
147:                 Err(_) => return Err("PUBLIC_HOST_KERNEL_COUNTERS".into()),
148:             };
149:             if count == 0 {
150:                 return std::str::from_utf8(&bytes[..used])
151:                     .map_err(|_| crate::Error::from("PUBLIC_HOST_KERNEL_COUNTERS"));
152:             }
153:             used += count;
154:             if used == bytes.len() {
155:                 return Err("PUBLIC_HOST_KERNEL_COUNTERS".into());
156:             }
157:         }
158:     }

160:     fn read_limit_text<'a>(
161:         reader: &mut impl Read,
162:         bytes: &'a mut [u8; MAX_LIMIT_BYTES + 1],
163:     ) -> Result<&'a str> {
164:         let mut used = 0;
165:         loop {
166:             // At least one byte remains until overflow is rejected. A short
167:             // read is not EOF, including at the original 128-byte boundary.
168:             let count = match reader.read(&mut bytes[used..]) {
169:                 Ok(count) => count,
170:                 Err(error) if error.kind() == io::ErrorKind::Interrupted => continue,
171:                 Err(_) => return Err("PUBLIC_HOST_KERNEL_UNAVAILABLE".into()),
172:             };
173:             if count == 0 {
174:                 return std::str::from_utf8(&bytes[..used])
175:                     .map_err(|_| crate::Error::from("PUBLIC_HOST_KERNEL_UNAVAILABLE"));
176:             }
177:             used += count;
178:             if used > MAX_LIMIT_BYTES {
179:                 return Err("PUBLIC_HOST_KERNEL_UNAVAILABLE".into());
180:             }
181:         }
182:     }

221:         // Resolve the actual directory on EVERY check. O_PATH preserves the

236:     fn read_limit_at<'a>(
237:         parent: &fs::File,
238:         name: &CStr,
239:         bytes: &'a mut [u8; MAX_LIMIT_BYTES + 1],
240:     ) -> Result<&'a str> {
241:         use rustix::fs::{openat, Mode, OFlags};
242:         let fd = loop {
243:             match openat(
244:                 parent,
245:                 name,
246:                 OFlags::RDONLY | OFlags::NOFOLLOW | OFlags::CLOEXEC | OFlags::NONBLOCK,
247:                 Mode::empty(),
248:             ) {
249:                 Err(error) if error == rustix::io::Errno::INTR => continue,
250:                 result => break result,
251:             }
252:         }
253:         .map_err(|_| crate::Error::from("PUBLIC_HOST_KERNEL_UNAVAILABLE"))?;
254:         let mut file = fs::File::from(fd);
255:         // These cgroup-v2 children are regular files. A substituted FIFO
256:         // must neither block nor lend buffered text or writer-free EOF.
257:         if !file
258:             .metadata()
259:             .map_err(|_| crate::Error::from("PUBLIC_HOST_KERNEL_UNAVAILABLE"))?
260:             .file_type()
261:             .is_file()
262:         {
263:             return Err("PUBLIC_HOST_KERNEL_UNAVAILABLE".into());
264:         }
265:         let text = read_limit_text(&mut file, bytes)?;
266:         #[cfg(test)]
267:         relative_tests::after_limit_read(name);
268:         Ok(text)
269:     }

295:             let current = file

315:                 && (metadata.dev(), metadata.ino(), metadata.uid()) == identity

389:     fn sample_counters(
390:         check: impl Fn() -> Result<ScopedParentDirectory>,
391:     ) -> Result<KernelResourceSample> {
392:         // The initial full check freshly opens and validates the parent. Only
393:         // this sample uses that local descriptor, which is dropped before the
394:         // final full check freshly opens the parent again. No FD or verdict is
395:         // retained across calls. Read/parse failures keep their earlier error.
396:         let sample = {
397:             let current = check()?;
398:             current.read_counters()?
399:         };
400:         check()?;
401:         Ok(sample)
402:     }

404:     // This is a common-case stack size, NOT a new limit on boot or membership
405:     // text. Longer input continues from the same stream on the original heap
406:     // path. Unlike read_to_string(File), the short path needs no size metadata.
407:     const CONTEXT_STACK_BYTES: usize = 256;
408: 
409:     fn read_context_text<'a>(
410:         reader: &mut impl Read,
411:         bytes: &'a mut [u8; CONTEXT_STACK_BYTES + 1],
412:     ) -> io::Result<Cow<'a, str>> {
413:         let mut used = 0;
414:         loop {
415:             let count = match reader.read(&mut bytes[used..]) {
416:                 Ok(count) => count,
417:                 Err(error) if error.kind() == io::ErrorKind::Interrupted => continue,
418:                 Err(error) => return Err(error),
419:             };
420:             if count == 0 {
421:                 return std::str::from_utf8(&bytes[..used])
422:                     .map(Cow::Borrowed)
423:                     .map_err(|error| io::Error::new(io::ErrorKind::InvalidData, error));
424:             }
425:             used += count;
426:             if used > CONTEXT_STACK_BYTES {
427:                 let mut complete = bytes[..used].to_vec();
428:                 // Preserve the already consumed prefix and read errors before
429:                 // UTF-8 validation, even when the prefix is malformed. Never
430:                 // reopen here: that could combine two different live files.
431:                 reader.read_to_end(&mut complete)?;
432:                 return String::from_utf8(complete)
433:                     .map(Cow::Owned)
434:                     .map_err(|error| io::Error::new(io::ErrorKind::InvalidData, error));
435:             }
436:         }
437:     }

439:     fn read_context<'a>(
440:         path: &Path,
441:         bytes: &'a mut [u8; CONTEXT_STACK_BYTES + 1],
442:     ) -> Result<Cow<'a, str>> {
443:         // A fresh open and complete read on EVERY check, including the current
444:         // calling thread's procfs path. Neither a descriptor nor its contents
445:         // are retained. I/O and UTF-8 errors keep the original context error.
446:         let mut file =
447:             fs::File::open(path).map_err(|_| crate::Error::from("PUBLIC_HOST_KERNEL_CONTEXT"))?;
448:         read_context_text(&mut file, bytes)
449:             .map_err(|_| crate::Error::from("PUBLIC_HOST_KERNEL_CONTEXT"))
450:     }

452:     fn in_parent_cgroup(parent: &str) -> Result<bool> {
453:         // A process may have independently placed native worker threads in a
454:         // cgroup-v2 threaded subtree. /proc/self identifies the thread-group
455:         // leader and cannot attest the actual calling worker's CPU cgroup.
456:         // The kernel's /proc/thread-self resolves to the current TID.
457:         let mut bytes = [0; CONTEXT_STACK_BYTES + 1];
458:         let text = read_context(Path::new("/proc/thread-self/cgroup"), &mut bytes)?;
459:         cgroup_within_parent(&text, parent)
460:     }

462:     fn cgroup_within_parent(text: &str, parent: &str) -> Result<bool> {
463:         let mut entries = text.lines();
464:         let only = entries.next().ok_or("PUBLIC_HOST_KERNEL_CONTEXT")?;
465:         if entries.next().is_some() {
466:             return Err("PUBLIC_HOST_KERNEL_CONTEXT".into());
467:         }
468:         let relative = only
469:             .strip_prefix("0::/")
470:             .ok_or("PUBLIC_HOST_KERNEL_CONTEXT")?;
471:         if relative.is_empty()
472:             || relative
473:                 .split('/')
474:                 .any(|x| x.is_empty() || x == "." || x == "..")
475:         {
476:             return Err("PUBLIC_HOST_KERNEL_CONTEXT".into());
477:         }
478:         Ok(within_parent(relative, parent))
479:     }

481:     fn within_parent(current: &str, parent: &str) -> bool {
482:         current == parent
483:             || current
484:                 .strip_prefix(parent)
485:                 .is_some_and(|suffix| suffix.starts_with('/'))
486:     }

488:     /// Accept additional future kernel cpu.stat keys, but never fabricate the
489:     /// four required counters, accept duplicates or accept malformed/huge input.
490:     /// The actual sysfs files are read only after the fixed parent was checked.
491:     fn resource_sample(cpu: &str, memory: &str, tasks: &str) -> Result<KernelResourceSample> {
492:         const INVALID: &str = "PUBLIC_HOST_KERNEL_COUNTERS";
493:         if cpu.len() > MAX_CPU_COUNTER_BYTES
494:             || memory.len() > MAX_SCALAR_COUNTER_BYTES
495:             || tasks.len() > MAX_SCALAR_COUNTER_BYTES
496:         {
497:             return Err(INVALID.into());
498:         }
499:         let mut fields = std::collections::BTreeMap::<&str, u64>::new();
500:         for line in cpu.lines() {
501:             let mut words = line.split_ascii_whitespace();
502:             let key = words.next().ok_or(INVALID)?;
503:             let raw = words.next().ok_or(INVALID)?;
504:             if words.next().is_some() {
505:                 return Err(INVALID.into());
506:             }
507:             let value = raw
508:                 .parse::<u64>()
509:                 .map_err(|_| crate::Error::from(INVALID))?;
510:             if fields.insert(key, value).is_some() {
511:                 return Err(INVALID.into());
512:             }
513:         }
514:         let required = |key| {
515:             fields
516:                 .get(key)
517:                 .copied()
518:                 .ok_or_else(|| crate::Error::from(INVALID))
519:         };
520:         let current = |raw: &str| -> Result<u64> {
521:             let mut parts = raw.split_ascii_whitespace();
522:             let value = parts
523:                 .next()
524:                 .ok_or(INVALID)?
525:                 .parse::<u64>()
526:                 .map_err(|_| crate::Error::from(INVALID))?;
527:             if parts.next().is_some() {
528:                 return Err(INVALID.into());
529:             }
530:             Ok(value)
531:         };
532:         Ok(KernelResourceSample {
533:             cpu_usage_usec: required("usage_usec")?,
534:             cpu_nr_periods: required("nr_periods")?,
535:             cpu_nr_throttled: required("nr_throttled")?,
536:             cpu_throttled_usec: required("throttled_usec")?,
537:             memory_current_bytes: current(memory)?,
538:             pids_current: current(tasks)?,
539:         })
540:     }
